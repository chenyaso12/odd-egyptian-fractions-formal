def OddEgyptian.UniformMinLengthTarget : Prop :=
∃ C,
  0 < C ∧
    ∀ (a b : ℕ),
      3 ≤ b →
        Odd b →
          1 ≤ a →
            a < b →
              (∃ s, OddEgyptian.Represents (↑a / ↑b) s ∧ s.card = OddEgyptian.oddMinLength a b) ∧
                ↑(OddEgyptian.oddMinLength a b) ≤ C * (1 + Real.log (Real.log ↑b))
